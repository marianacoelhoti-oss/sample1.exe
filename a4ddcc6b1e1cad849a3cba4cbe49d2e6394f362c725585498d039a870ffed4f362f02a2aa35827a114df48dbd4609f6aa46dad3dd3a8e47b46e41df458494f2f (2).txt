; Demonstração educacional de técnicas de malware
; APENAS PARA FINS EDUCACIONAIS
section .data
    ; String ofuscada (XOR com a chave 0x41) - "Unicesumar\n"
    encoded_msg db 0x14, 0x2F, 0x28, 0x22, 0x24, 0x32, 0x34, 0x2C, 0x20, 0x33, 0x4B
    ; Decodificada: "Unicesumar\n"
    msg_len equ $ - encoded_msg

    ptrace_fail_msg db 'Debugger detectado - saindo...', 0xA
    ptrace_fail_len equ $ - ptrace_fail_msg

    running_msg db 'Executando normalmente...', 0xA
    running_len equ $ - running_msg

section .bss
    decoded_buffer resb 64     ; Buffer para a mensagem decodificada

section .text
    global main

main:
    ; 1. Verificar debugger usando ptrace anti-debugging
    call check_debugger

    ; 2. Decodificar a mensagem ofuscada
    call decode_message

    ; 3. Exibir a mensagem decodificada
    mov eax, 4                 ; syscall write
    mov ebx, 1                 ; stdout
    mov ecx, decoded_buffer    ; mensagem decodificada
    mov edx, msg_len           ; comprimento
    int 0x80                   ; chamada ao kernel

    ; 4. Finalizar o programa
    jmp exit_program

; --------------------------------------------
; Função para verificar se estamos em debugger
; --------------------------------------------
check_debugger:
    ; Usar ptrace para detectar debugger
    ; Se PTRACE_TRACEME falhar, um debugger já está anexado
    mov eax, 26                ; syscall ptrace
    mov ebx, 0                 ; PTRACE_TRACEME
    mov ecx, 0
    mov edx, 0
    mov esi, 0
    int 0x80

    ; Verificar resultado (eax < 0 significa erro)
    cmp eax, 0
    jge debugger_not_found

    ; Debugger detectado - exibir mensagem e sair
    mov eax, 4                 ; syscall write
    mov ebx, 1                 ; stdout
    mov ecx, ptrace_fail_msg   ; mensagem
    mov edx, ptrace_fail_len   ; comprimento
    int 0x80

    ; Sair do programa
    jmp exit_program

debugger_not_found:
    ; Exibir mensagem de execução normal
    mov eax, 4                 ; syscall write
    mov ebx, 1                 ; stdout
    mov ecx, running_msg       ; mensagem
    mov edx, running_len       ; comprimento
    int 0x80
    ret

; --------------------------------------------
; Função para decodificar a mensagem
; --------------------------------------------
decode_message:
    mov ecx, msg_len           ; Contador para loop
    mov esi, 0                 ; Índice para string

decode_loop:
    mov al, byte [encoded_msg + esi]  ; Obter byte da string codificada
    xor al, 0x41                      ; Decodificar com XOR 0x41
    mov [decoded_buffer + esi], al    ; Armazenar resultado no buffer
    inc esi                           ; Incrementar índice
    dec ecx                           ; Decrementar contador
    jnz decode_loop                   ; Continuar se não for zero
    ret

; --------------------------------------------
; Função para finalizar o programa
; --------------------------------------------
exit_program:
    mov eax, 1                 ; syscall exit
    xor ebx, ebx               ; código de retorno 0
    int 0x80
